Private Sub AdaugaPrezenta0()
If Not IsLoaded("Prezenta") Then Exit Sub

Dim i As Long
Dim Dt As Date
Dim WrkDayz As Integer
Dim MaxIDZ As Double

Forms!prezenta!IDL = IDL
Forms!prezenta!IDV = IDV
FOC.visible = True

Dim RC As DAO.Recordset, TC As DAO.Recordset

Set TC = CurrentDb.OpenRecordset("LunaD", 2)

TC.AddNew
    TC!IDL = IDL
    TC!IDV = IDV
    TC!Luna = Luna
    TC!Anul = Anul
    TC!LunaT = LunaT
    TC!LA = LA
TC.Update

Set RC = CurrentDb.OpenRecordset("SELECT Platitori.IDG, Platitori.IDP FROM Platitori WHERE Platitori.Plecat=False;", 2)
Set TC = CurrentDb.OpenRecordset("Prezenta", 2)

If Not POP(RC) Then
    MsgBox "A aparut o eroare la preluarea generala!" & vbCrLf & "Este necesar sa o realizezi manual!", vbOKOnly + vbExclamation
    GoTo IESIRE
End If

Dt = CDate(Format("01/" & Me.Luna & "/" & Me.Anul, "Short Date"))
WrkDayz = Work_Days(Dt, CDate(Format(dhLastDayInMonth(Dt), "Short Date")))
MaxIDZ = DMax("IDZ", "Prezenta")

For i = 1 To RC.RecordCount
    TC.AddNew
        TC!IDZ = MaxIDZ + i
        TC!IDG = RC!IDG
        TC!IDP = RC!IDP
        TC!IDL = IDL
        TC!IDV = IDV
        TC!ZileLuna = WrkDayz
        TC!ZilePrezenta = 0
        TC!ZileAbsenta = 0
    TC.Update
    
    Call ProgressBar(Me, RC.RecordCount, i)
    
    RC.MoveNext
Next i

IESIRE:
    Set RC = Nothing
    Set TC = Nothing
    
    Call ProgressBar(Me, -1, -1)
    
    Exit Sub
    
End Sub

Private Sub Form_Load()
SeDeschide = True
Me.AllowAdditions = True
Me.AllowEdits = True
Me.AllowDeletions = True
Anul.Enabled = False
Anul.Locked = True

Call ProgressBar(Forms!prezenta!Luni.Form, -1, -1)

Dim RC As Recordset, TC As Recordset

Set TC = CurrentDb.OpenRecordset("LunaD_L", 2)
Set RC = CurrentDb.OpenRecordset("SELECT lunad.IDL, lunad.Idv, lunad.Luna, lunad.Anul, lunad.LunaT, lunad.LA, lunad.inchisa FROM lunad;", 2)

Call GolesteTabel(TC)

Do While Not RC.EOF
    TC.AddNew
        TC!IDL = RC!IDL
        TC!Luna = RC!Luna
        TC!Anul = RC!Anul
        TC!LunaT = RC!LunaT
        TC!LA = RC!LA
        TC!IDV = RC!IDV
        TC!Inchisa = RC!Inchisa
    TC.Update
    RC.MoveNext
Loop

Set RC = Nothing
Set TC = Nothing

If Not Me.NewRecord Then
    FOC.visible = True
    FOC.SetFocus
End If

Me.Requery
SeDeschide = False

End Sub

Private Sub bINC_Click()
If Nz(DLookup("IDL", "LunaD", "IDL=" & Me.IDL), "") = "" Then
    MsgBox "Luna nu a fost activata inca, deci nu poate fi inchisa!", vbOKOnly + vbInformation
    Exit Sub
End If

If Not Inchisa Then
    If ECount("IDG", "(SELECT IDG FROM SS_Buget WHERE IDL = " & Me.IDL - 2 & " GROUP BY IDG)") <> 0 Then
        If ECount("IDG", "(SELECT IDG FROM SS_Buget WHERE IDL = " & Me.IDL - 2 & " GROUP BY IDG)") > ECount("IDG", "(SELECT IDG FROM Prezenta WHERE IDL = " & IDL & " GROUP BY IDG)") Then
            If MsgBox("In luna anterioara exista mai multe grupe cu prezenta decat in luna curenta!" & vbCrLf & "Este corect?", vbYesNo + vbQuestion) = vbNo Then Exit Sub
        End If
    End If
    
    If MsgBox("Doresti inchiderea lunii selectate?" & vbCrLf & "Nu vei mai putea modifica prezenta, dar vei putea modifica platile!" & vbCrLf & "Doresti sa continui?", vbYesNo + vbQuestion) = vbNo Then Exit Sub
    Dim RC As Recordset
    Dim tIDL As Integer
    
    Set RC = CurrentDb.OpenRecordset("SELECT * FROM LunaD WHERE IDL = " & IDL, 2)
    If Not RC.EOF Then
        tIDL = RC!IDL - 1
        
        RC.Edit
            RC!Inchisa = True
        RC.Update
        
        Set RC = CurrentDb.OpenRecordset("SELECT * FROM LunaD_L WHERE IDL = " & IDL, 2)
        RC.Edit
            RC!Inchisa = True
        RC.Update
    
        If tIDL <> 0 Then
            Screen.MousePointer = 11
            Call CalculSituatieDebitori_Buget2021(True, True, tIDL)
            Screen.MousePointer = 0
            Me.Recalc
        End If
        
    Else
        MsgBox "Luna selectata nu a putut fi inchisa!" & vbCrLf & "Contacteaza autorul!", vbOKOnly + vbCritical
        Me.Undo
        Luna.Undo
    End If
Else
    If DMax("IDL", "LunaD") = IDL Then
        If MsgBox("Doresti deblocarea lunii curente?", vbYesNo + vbQuestion) = vbNo Then Exit Sub
        ActualizeazaCamp "LunaD", "Inchisa", False, "IDL=" & IDL
        ActualizeazaCamp "LunaD_L", "Inchisa", False, "IDL=" & IDL
        GolesteTabel , "SS_Buget", "IDL=" & IDL - 1
        Me.Recalc
    Else
        MsgBox "Se poate debloca DOAR ultima luna inchisa, atata timp cat nu exista una deschisa!", vbOKOnly + vbInformation
    End If
End If
    
Set RC = Nothing

End Sub

Private Sub Luna_AfterUpdate()
Dim RC As DAO.Recordset

If DCount("IDL", "LunaD", "Inchisa=False") <> 0 Then
    MsgBox "Pentru a adauga o luna noua, luna anterioara trebuie sa fie inchisa!" & vbCrLf & "Apasa pe lacatul verde!", vbOKOnly + vbCritical

    Me.Requery
    Exit Sub
End If

Anul = IIf(Luna.Column(0) = "1", DMax("Anul", "LunaD") + 1, DMax("Anul", "LunaD"))
LunaT = Luna.Column(1)
LA = IIf(Len(Luna) = 1, "0" & Luna, Luna) & Anul

IDV = CDbl(Nz(DLookup("IDV", "ValoriTaxe", "Activ=True"), 0))

If DCount("IDL", "LunaD") <= 0 Then Exit Sub

If MsgBox("Doresti preluarea tuturor grupelor din luna " & IIf(Luna = 1, 12, Luna - 1) & "/" & IIf(Luna = 1, Year(Now()) - 1, Year(Now())) & "?", vbYesNo + vbQuestion) = vbNo Then Exit Sub

Call AdaugaPrezenta0

MsgBox "Datele au fost transferate cu succes!", vbOKOnly + vbInformation

FOC.SetFocus

End Sub

Sub TesteazaUltimaLuna()

Sub IncarcaGrupe()

Forms!prezenta!Grupe!GE.visible = True

CurrentDb.Execute "DELETE * FROM Grupe_L"

Dim RC As Recordset, TC As Recordset
Set TC = CurrentDb.OpenRecordset("Grupe_L", 2)

If Me!Inchisa = True Then
    Set RC = CurrentDb.OpenRecordset("SELECT IDG, Grupa, Educator FROM SS_Buget WHERE IDL = " & Me!IDL & " GROUP BY IDG, Grupa, Educator ORDER BY IDG;", 2)
Else
    Set RC = CurrentDb.OpenRecordset("SELECT Grupe.IDg, Grupe.Grupa, grupe.educator FROM Grupe;", 2)
End If
   
Do While Not RC.EOF
    TC.AddNew
        TC!IDG = RC!IDG
        TC!Grupa = RC!Grupa
        TC!Educator = RC!Educator
    TC.Update
    RC.MoveNext
Loop

Set RC = Nothing
Set TC = Nothing

Forms!prezenta!Grupe.Requery

End Sub
