Private Sub bPrel_Click()
Dim RC As DAO.Recordset
Dim i As Long
Dim Dt As Date
Dim WrkDayz As Integer
Dim MaxIDZ As Double

If Forms!prezenta!Inchisa = True Then
    MsgBox "Nu se pot prelua copii intr-o luna inchisa!", vbOKOnly + vbExclamation
    Exit Sub
End If

Set RC = CurrentDb.OpenRecordset("SELECT Platitori.IDP, Platitori.IDG FROM Platitori WHERE Platitori.IDP Not In (SELECT IDP FROM [Prezenta] WHERE [Prezenta].[IDL] = " & [Forms]![prezenta]![IDL] & " AND [Prezenta].[IDG]=" & Forms!prezenta!IDG & ") AND Platitori.IDG = " & [Forms]![prezenta]![IDG] & " AND Plecat = False", 2)
If Not POP(RC) Then
    MsgBox "Nu exista copii nepreluati pentru grupa selectata!", vbOKOnly + vbInformation
    Set RC = Nothing
    Exit Sub
End If

Dim TC As DAO.Recordset
Set TC = CurrentDb.OpenRecordset("Prezenta", 2)

Dt = CDate(Format("01/" & Forms!prezenta!Luna & "/" & Forms!prezenta!Anul, "Short Date"))
WrkDayz = Work_Days(Dt, CDate(Format(dhLastDayInMonth(Dt), "Short Date")))
MaxIDZ = DMax("IDZ", "Prezenta")

For i = 1 To RC.RecordCount
    TC.AddNew
        TC!IDZ = MaxIDZ + i
        TC!IDG = RC!IDG
        TC!IDP = RC!IDP
        TC!IDL = Forms!prezenta!IDL
        TC!IDV = Forms!prezenta!IDV
        TC!ZileLuna = WrkDayz
        TC!ZilePrezenta = 0
        TC!ZileAbsenta = 0
    TC.Update
    
    RC.MoveNext
Next i

Set RC = Nothing
Set TC = Nothing

Call IncarcaPrezenta(True)

MsgBox "Toti copiii nou adaugati au fost preluati pentru grupa selectata!", vbOKOnly + vbInformation

Private Sub Sav_Click()
If Forms!prezenta!Inchisa = True Then
    MsgBox "Nu se pot salva date intr-o luna inchisa!", vbOKOnly + vbExclamation
    Exit Sub
End If

If MsgBox("Doresti salvarea datelor pentru luna " & Forms!prezenta!Luna & "/" & Forms!prezenta!Anul & " si grupa " & Forms!prezenta!Grupe!Grupa & "?", vbYesNo + vbQuestion) = vbNo Then Exit Sub

On Error GoTo Eroare

If Me.Dirty Then Me.Dirty = False

Dim Wk As DAO.Workspace
Dim i As Integer

Set Wk = DBEngine.Workspaces(0)

Dim RC As Recordset, TC As Recordset

Wk.BeginTrans

Set RC = CurrentDb.OpenRecordset("SELECT * FROM LunaD_L WHERE IDL NOT IN (SELECT IDL FROM LunaD) AND IDL IN (SELECT IDL FROM Prezenta_L)", 2)
POP RC

If RC.EOF Then 'nu s-au adaugat luni noi
    GoTo FaraLunaNoua
End If

Set TC = CurrentDb.OpenRecordset("LunaD", 2)
Do While Not RC.EOF
    TC.AddNew
        TC!IDL = RC!IDL
        TC!IDV = Parent!IDV
        TC!Luna = RC!Luna
        TC!Anul = RC!Anul
        TC!LunaT = RC!LunaT
        TC!LA = RC!LA
    TC.Update
    RC.MoveNext
Loop

FaraLunaNoua:

Call GolesteTabel(, "Prezenta_L", "IDZ IN (SELECT IDZ FROM Prezenta_L WHERE Sters=True)") 'elimin copii stersi din temporar

Set RC = CurrentDb.OpenRecordset("SELECT * FROM Prezenta_L", 2)
If RC.EOF Then 'ciudat
    GoTo FaraPrezenta
End If

Do While Not RC.EOF
    Set TC = CurrentDb.OpenRecordset("SELECT * FROM Prezenta WHERE IDP=" & RC!IDP & " AND IDL=" & RC!IDL & " AND IDG=" & RC!IDG, 2)
    If Not POP(TC) Then
        TC.AddNew
    Else
        TC.Edit
    End If
    
        TC!IDZ = RC!IDZ
        TC!IDP = RC!IDP
        TC!IDL = RC!IDL
        TC!IDV = Parent!IDV
        TC!IDG = RC!IDG
        TC!ZileLuna = RC!ZileLuna
        TC!ZilePrezenta = RC!ZilePrezenta
        TC!ZileAbsenta = RC!ZileAbsenta
        TC!MZCPrezenta = RC!MZCPrezenta
        TC!MZCAbsenta = RC!MZCAbsenta
        TC!ValoareContract = RC!ValoareContract
        TC!ValoareMancare = RC!ValoareMancare
        TC!ValoareTotala = RC!ValoareTotala
        TC!ReducereFrate = RC!ReducereFrate
        TC!ReducereBonus = RC!ReducereBonus
        TC!SI = RC!SI
        TC!Detalii = RC!Detalii
        TC!Avans = RC!Avans
        TC!Restanta = RC!Restanta
    TC.Update
    RC.MoveNext
Loop

FaraPrezenta:

Wk.CommitTrans

MsgBox "Salvare efectuata cu succes!", vbOKOnly + vbInformation

Set RC = Forms!prezenta!Luni.Form.Recordset
    RC.FindFirst "IDL = " & Forms!prezenta!IDL

Set RC = Forms!prezenta!Grupe.Form.Recordset
    RC.FindFirst "IDG = " & Forms!prezenta!IDG
    
IESIRE:
Set RC = Nothing
Set TC = Nothing
Set Wk = Nothing

Exit Sub

Eroare:
    Wk.Rollback
    MsgBox "Eroare salvare luna: " & vbCrLf & Err.Number & vbCrLf & Err.Description
    Resume IESIRE
End Sub
