Private Sub xForm_Open(Cancel As Integer)
SeDeschide = True

    GE.visible = True
    
    Dim RC As Recordset, TC As Recordset
    Set TC = CurrentDb.OpenRecordset("Grupe_L", 2)
    Set RC = CurrentDb.OpenRecordset("SELECT Grupe.IDg, Grupe.Grupa, grupe.educator FROM Grupe;", 2)
    
    Call GolesteTabel(TC)
    
    Do While Not RC.EOF
        TC.AddNew
            TC!IDG = RC!IDG
            TC!Grupa = RC!Grupa
            TC!Educator = RC!Educator
        TC.Update
        RC.MoveNext
    Loop
    
    Set RC = Nothing
    Set TC = Nothing
    
    
    Me.AllowAdditions = False
    Me.AllowEdits = False
    
    Me.Requery


SeDeschide = False
End Sub
