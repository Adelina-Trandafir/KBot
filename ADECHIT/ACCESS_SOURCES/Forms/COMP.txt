Private Sub BTN_OK_Click()
With CurrentDb.OpenRecordset("COMP")
    .AddNew
    !IDP_D = COMP_D!D_IDP
    !IDP_C = COMP_C!C_IDP
    !Suma = Suma
    .Update
End With

COMP_D.Requery
COMP_C.Requery

End Sub

Private Sub Form_Load()
Load
End Sub

Private Sub Load()
CurrentDb.Execute "DELETE * FROM tmpCOMP_D"
CurrentDb.Execute "DELETE * FROM tmpCOMP_C"
'CurrentDb.Execute "DELETE * FROM tmpCOMP"

Dim LIDL As Long
Dim SQL As String

SQL = CurrentDb.QueryDefs("qCOMP_D").SQL

LIDL = CurrentDb.OpenRecordset("SELECT MAX(IDL) FROM Prezenta").Fields(0).value

SQL = CurrentDb.QueryDefs("qCOMP_D").SQL

SQL = Replace(SQL, "[LIDL]", LIDL)

CurrentDb.Execute "INSERT INTO tmpCOMP_D (D_IDP, D_IDG, D_GRUPA, D_NUME, D_SOLD, D_REST) " & SQL
                    
CurrentDb.Execute "INSERT INTO tmpCOMP_C (C_IDP, C_IDG, C_GRUPA, C_NUME, C_SOLD, C_REST) " & _
                    "SELECT IDP, IDG, GRUPA, NUME, SFC, SFC FROM SS_Buget " & _
                    "WHERE ROUND(SFC,2) > 0 AND IDL = " & LIDL - 1 & " " & _
                    "ORDER BY GRUPA, NUME"
                    
COMP_D.Requery
COMP_C.Requery
End Sub

Private Sub SUMA_BeforeUpdate(Cancel As Integer)
If Round(Suma) > Round(C_SOLD) Then
    MsgBox "NU EXISTA SOLD SUFICIENT PENTRU " & C_NUME & "!", vbCritical
    Cancel = True
    SendKeysAPI "{ESC}"
    SendKeysAPI "{ESC}"
End If
End Sub
