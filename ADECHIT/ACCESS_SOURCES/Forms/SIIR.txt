Private Sub AlegeCale_Click()
Dim FD As FileDialog
Dim fileNamePath As String, fileExtension As String, FileName As String

GolesteTabel , "IMPXLS"

If fileNamePath = "" Then
    Set FD = Application.FileDialog(msoFileDialogOpen)
    Dim FileChosen As Integer
    FileChosen = FD.Show
    FD.Title = "Choose workbook"
    FD.InitialView = msoFileDialogViewList

    FD.Filters.Clear
    FD.Filters.Add "Fisiere Excel < 2007", "*.xls"
    FD.Filters.Add "Fisiere Excel > 2007", "*.xlsx"
    'FD.Filters.Add "All files", "*.*"
    FD.FilterIndex = 1
    FD.ButtonName = "Alege fisier"
    If FileChosen <> -1 Then 'didn't choose anything (clicked on CANCEL)
        MsgBox "Nu ai ales niciu fisier!", vbCritical
    Else
        fileNamePath = FD.SelectedItems(1)
        Cale = fileNamePath
        FileName = Dir(fileNamePath)
        fileExtension = Right$(FileName, Len(FileName) - InStrRev(FileName, "."))
        ImportaDateExcel
    End If
    Set FD = Nothing
End If

End Sub

Private Sub btnIMP_Click()
On Error GoTo Eroare

Dim RC As DAO.Recordset
Dim TC As DAO.Recordset
Dim Wk As DAO.Workspace

Dim i As Long

Dim IDP As Integer, IDG As Integer

If Nz(Grupa, "") = "" Then
    MsgBox "Nu ai introdus grupa!", vbOKOnly + vbCritical
    Exit Sub
End If

If Nz(E1, "") = "" Then
    MsgBox "Nu ai introdus numele educatorului (1)!", vbOKOnly + vbCritical
    Exit Sub
End If

If Nz(IDV, "") = "" Then
    MsgBox "Nu ai selectat tipul programului (normal / prelungit)!", vbOKOnly + vbCritical
    Exit Sub
End If

Set Wk = Workspaces(0)

Set RC = CurrentDb.OpenRecordset("SELECT * FROM IMPXLS ORDER BY ID", 2)

If MsgBox("Doresti preluarea celor " & DCount("ID", "IMPXLS", "S=True") & " copii selectati aferenti grupei " & UCase(Grupa) & "?", vbYesNo + vbQuestion) = vbNo Then GoTo IESIRE

Wk.BeginTrans

IDP = CDbl(Nz(DMax("IDP", "Platitori"), 0)) + 1

If Nz(DLookup("IDG", "Grupe", "UCase(Grupa) Like '" & UCase(Grupa) & "'"), "") <> "" Then
    MsgBox "Grupa " & UCase(Grupa) & " exista in baza de date!" & vbCrLf & "Nu se poate continua cu preluarea datelor!", vbOKOnly + vbCritical
    GoTo IESIRE
Else
    Set TC = CurrentDb.OpenRecordset("Grupe", 2)
        TC.AddNew
            TC!IDG = CInt(Nz(DMax("IDG", "Grupe"), 0)) + 1
            IDG = TC!IDG
            TC!Grupa = Grupa
            TC!Educator = Nz(E1, "") & IIf(Nz(E2, "") <> "", " | " & E2, "")
        TC.Update
End If

Set TC = CurrentDb.OpenRecordset("Platitori", 2)
Do While Not RC.EOF
    i = i + 1
    TC.AddNew
        TC!IDP = RC!IDP
        TC!IDG = IDG
        TC!IDV = IDV
        TC!Nume = RC!numeprenume
        TC!Adresa = Nz(RC!Adresa, "#FARA ADRESA#")
        TC!SI = RC!SI
        TC!CNP = RC!CNP
    TC.Update
    
    RC.MoveNext
Loop

RC.MoveFirst

Set TC = CurrentDb.OpenRecordset("Platitori_sub", 2)
Do While Not RC.EOF
    TC.AddNew
        TC!IDS = RC!IDP
        TC!IDP = RC!IDP
        TC!Nume = RC!numeprenume
        TC!Adresa = Nz(RC!Adresa, "#FARA ADRESA#")
        TC!Activ = True
    TC.Update
    
    RC.MoveNext
Loop

RC.MoveFirst

Set TC = CurrentDb.OpenRecordset("Delegati", 2)
Do While Not RC.EOF
    TC.AddNew
        TC!IDD = RC!IDP
        TC!IDS = RC!IDP
        TC!Nume = RC!numeprenume
        TC!CNP = RC!CNP
    TC.Update
    
    RC.MoveNext
Loop

Wk.CommitTrans

MsgBox "Import efectuat cu succes!", vbOKOnly + vbInformation

IESIRE:

Set Wk = Nothing
Set RC = Nothing
Set TC = Nothing

Exit Sub

Eroare:
    DebugPrint Err.Description
    Wk.Rollback
    Resume IESIRE
        
End Sub

Sub ImportaDateExcel()
Dim xlApp As Object
Dim xlSheet As Object
Dim RC As DAO.Recordset
Dim i As Integer

Set xlApp = CreateObject("Excel.Application")
Set xlSheet = xlApp.Workbooks.Open(Cale).Sheets(1)
xlApp.visible = False

If Not xlSheet Is Nothing Then
    Set RC = CurrentDb.OpenRecordset("IMPXLS", 2)
Else
    Exit Sub
End If

Grupa = xlSheet.Range("A1").value
E1 = xlSheet.Range("G1").value
E2 = xlSheet.Range("G2").value

For i = 7 To 50
    If xlSheet.Range("A" & i).value = "" Then GoTo NextI
    
    RC.AddNew
        RC!CNP = Trim(xlSheet.Range("A" & i).value)
        RC!Nume = xlSheet.Range("B" & i).value
        RC!it = xlSheet.Range("C" & i).value
        RC!prenume = xlSheet.Range("D" & i).value
        RC!prenume_2 = xlSheet.Range("E" & i).value
        RC!prenume_3 = xlSheet.Range("F" & i).value
        RC!numeprenume = Trim(Trim(RC!Nume) & " " & Trim(RC!it & IIf(RC!it = "-", "", ".")) & " " & Trim(RC!prenume) & " " & Trim(RC!prenume_2) & " " & Trim(RC!prenume_3))
        RC!SI = 0
        RC!S = True
        RC!IDP = CDbl(Nz(DMax("IDP", "Platitori"), 0)) + i - 6
    RC.Update
    ProgressBar Me, 43, i - 6, False
NextI:
Next

xlApp.Quit

ProgressBar Me, -1, 0

Set xlApp = Nothing
Set xlSheet = Nothing

IMPXLS.SourceObject = "IMPXLS"

    

End Sub
