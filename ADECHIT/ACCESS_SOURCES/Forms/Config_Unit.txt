Private Sub CodAnulare_BeforeUpdate(Cancel As Integer)
Dim Cod As String
Cod = InputBox("Completeaza codul!" & vbCrLf & "Valoare 1: " & DMax("Numar", "Chitante") & vbCrLf & "Valoare 2: " & Format(Now(), "Short Date"), "Schimbare mod anulare documente...", "")
If Cod <> MakeCod(DMax("Numar", "Chitante"), Format(Now(), "Short Date")) Then
    MsgBox "Codul nu corespunde!"
    Cancel = True
    Me.Undo
End If

End Sub

Private Sub EDC_Click()
If DMax("Year(Data)", "Chitante", "Anulata=False") = Year(Now()) Then
    MsgBox "Exista Chitante salvate pe anul curent! Nu se mai poate modifica numarul acestora!", vbOKOnly + vbCritical
    Exit Sub
End If
Dim msg
msg = InputBoxDK("Tasteaza parola pentru modificarea numarului chitantelor!", "CLng(Date)", "")
If msg = "" Then
    Exit Sub
ElseIf msg <> CLng(Date) Then Exit Sub
End If

Numar.Enabled = True
Numar.Locked = False
Numar.SetFocus

End Sub

Private Sub Form_Load()
'informatii unitate
Dim i As Integer
Dim RC As DAO.Recordset: Set RC = CurrentDb.OpenRecordset("Unitati")

For i = 0 To RC.Fields.Count - 1
    If ControlExists(RC.Fields(i).Name, Me) Then Me.Controls("" & RC.Fields(i).Name & "") = RC.Fields(i).value
Next i

Set RC = CurrentDb.OpenRecordset("SELECT * FROM CFGs WHERE F='BF'", dbOpenDynaset)
POP RC
Do While Not RC.EOF
    Me.Controls("" & RC!CFG & "").value = RC!Vl
    RC.MoveNext
Loop

Set RC = CurrentDb.OpenRecordset("SELECT * FROM CFGs WHERE F='CH'", dbOpenDynaset)
POP RC
Do While Not RC.EOF
    Me.Controls("" & RC!CFG & "").value = RC!Vl
    RC.MoveNext
Loop

Set RC = CurrentDb.OpenRecordset("SELECT * FROM CFGs WHERE F='XX'", dbOpenDynaset)
POP RC
Do While Not RC.EOF
    Me.Controls("" & RC!CFG & "").value = RC!Vl
    RC.MoveNext
Loop

IESIRE.SetFocus

If DLookup("VL", "CFGs", "CFG='Platitor'") = "Buget" Then
    P1.visible = False
    P2.visible = False
Else
    P1.visible = True
    P2.visible = True
End If

NouFirma = False

Set RC = Nothing


End Sub

Private Sub Salvare_Click()
Dim RC As Recordset: Set RC = CurrentDb.OpenRecordset("SELECT * FROM Unitati WHERE IdUnitate=" & IdUnitate)
Dim i As Integer
Dim ctl As Control

If MsgBox("Doresti salvarea modificarilor efectuate?", vbYesNo + vbQuestion) = vbNo Then Exit Sub
If NouFirma Then RC.AddNew Else RC.Edit
'    rc!IdUnitate = IdUnitate
    RC!Denumire = Denumire
    RC!CF = CF
    RC!J = J
    RC!Adresa = Adresa
    RC!Judetul = Judetul
    RC!Cont = Cont
    RC!Banca = Banca
    RC!F_SERIE = F_SERIE
    RC!F_NUMAR = F_NUMAR
    RC!F_CONTINUT = F_CONTINUT
    RC!F_R1 = F_R1
    RC!F_R2 = F_R2
RC.Update

Set RC = CurrentDb.OpenRecordset("SELECT * FROM CFGs WHERE F='BF'")

For Each ctl In Me.Controls
    If ctl.Properties(2) = 109 Or ctl.Properties(2) = 106 Then
        RC.FindFirst "CFG='" & ctl.Name & "'"
        If Not RC.NoMatch Then
            RC.Edit
                RC!Vl = ctl.value
            RC.Update
        End If
    End If
Next

Set RC = CurrentDb.OpenRecordset("SELECT * FROM CFGs WHERE F='CH'")

For Each ctl In Me.Controls
    If ctl.Properties(2) = 109 Or ctl.Properties(2) = 106 Then
        RC.FindFirst "CFG='" & ctl.Name & "'"
        If Not RC.NoMatch Then
            RC.Edit
                RC!Vl = ctl.value
            RC.Update
        End If
    End If
Next

Set RC = CurrentDb.OpenRecordset("SELECT * FROM CFGs WHERE F='XX'")

For Each ctl In Me.Controls
    If ctl.Properties(2) = 109 Or ctl.Properties(2) = 106 Then
        RC.FindFirst "CFG='" & ctl.Name & "'"
        If Not RC.NoMatch Then
            RC.Edit
                RC!Vl = ctl.value
            RC.Update
        End If
    End If
Next

Set RC = Nothing
Set RC = Nothing
Set ctl = Nothing

MsgBox "Salvare efectuata cu succes!", vbOKOnly + vbInformation
End Sub
