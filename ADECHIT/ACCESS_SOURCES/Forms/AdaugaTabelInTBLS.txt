Private Sub Command2_Click()
Dim Tbl As TableDef
Dim Bz As String

For Each Tbl In CurrentDb.TableDefs
    If Tbl.Name = L Then
        Bz = L.Column(5)
        Bz = Replace(Bz, "DATABASE=" & CurrentProject.path & "\", "")
        Bz = Replace(Bz, ".mdb", "")
        
        Call AdaugTabelIn_TBLS( _
                    "#", _
                    Tbl, _
                    Nz(DMax("Struct", "_TBLS_"), 0) + 1, _
                    Bz)
    End If
Next

For i = 1 To L.ListCount
    L.RemoveItem 0
Next

Call Form_Load

End Sub

Private Sub Form_Load()
Dim Tbl As TableDef
Dim x As String, i As Integer




For Each Tbl In CurrentDb.TableDefs
    For i = 0 To 5
        x = x & Tbl.Properties(i).Name & ";"
    Next
    x = Mid(x, 1, Len(x) - 1)
    L.AddItem x
    Exit For
Next

x = ""

For Each Tbl In CurrentDb.TableDefs
    If Nz(Tbl.Properties(4).value, "") = "" Then GoTo Pas
    If Nz(DLookup("Struct", "_TBLS_", "NumTbl='" & Tbl.Properties(0) & "'"), "") <> "" Then GoTo Pas
    For i = 0 To 5
        x = x & Tbl.Properties(i).value & ";"
    Next
    x = Mid(x, 1, Len(x) - 1)
    TblProps Tbl
    ''debugprint X
    L.AddItem x
Pas:
    x = ""
Next

End Sub

Function TblProps(Tbl As TableDef)
Dim i As Integer
Open CurrentProject.path & "\tblprops.txt" For Append As #1
For i = 0 To 9
    Print #1, Tbl.Name; "--> (" & i & ") " & Tbl.Properties(i).Name & " = " & Tbl.Properties(i).value
Next i
'Print #1, vbCrLf
Close #1
End Function

Private Sub L_DblClick(Cancel As Integer)
Call Command2_Click
