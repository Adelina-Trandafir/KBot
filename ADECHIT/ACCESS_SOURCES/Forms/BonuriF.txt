Private Sub Listeaza_Click()

    MsgBox "Completeaza toate campurile!", vbOKOnly + vbCritical
    Exit Sub

If MsgBox("Pentru listarea bonului este necesara salvarea datelor!" & vbCrLf & "Doresti sa continui?", vbYesNo + vbQuestion) = vbNo Then Exit Sub

Dim RC As Recordset
Set RC = CurrentDb.OpenRecordset("SELECT * FROM BonuriF WHERE IDPL=" & IDPL, 2)

If RC.EOF Then
    RC.AddNew
Else
    RC.Edit
End If
    RC!IDB = IDB
    RC!IDPL = IDPL
    RC!NrBF = NrBF
    RC!Valoare = Valoare
    RC!Continut = Continut
RC.Update

Set RC = CurrentDb.OpenRecordset("SELECT * FROM BonuriF WHERE IDPL=" & IDPL, 2)
    RC.Edit
    RC!Mesaj = fBonFiscal(IDPL)
    If RC!Mesaj <> "FINAL" Then
        MsgBox "A aparut o eroare la comunicarea cu casa de marcat!" & vbCrLf & "Incearca din nou!", vbOKOnly + vbInformation
        Listat = False
        RC!Listat = False
        RC!Corect = False
    Else
        If MsgBox("Bonul a fost listat corect?", vbYesNo + vbQuestion) = vbYes Then
            RC!Corect = True
            RC!Listat = True
            DataBF = Format(Now(), "Short Date")
            RC!DataBF = DataBF
            OraBF = Format(Now(), "Long Time")
            RC!OraBF = OraBF
        Else
            RC!Listat = True
            RC!Corect = False
            MsgBox "In cazul in care casa este pornita si legata corect la calculator, sun-o pe Adelina. Nu se supara! (0721201690)!", vbOKOnly + vbQuestion
        End If
    End If
RC.Update

If RC.EOF Then
    RC.AddNew

    RC.Edit

    RC!IDB = IDB
