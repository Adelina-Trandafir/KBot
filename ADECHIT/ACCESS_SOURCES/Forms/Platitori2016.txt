Private Sub Form_Open(Cancel As Integer)
Call FRMOPN

'Exit Sub

Dim RC As Recordset, TC As Recordset
Dim i As Integer

'======== actualizare tabel [Delegati] daca nu s-a creeat rand la introducerea copilului
Set RC = CurrentDb.OpenRecordset("SELECT Platitori_sub.* FROM Platitori_sub WHERE (((Platitori_sub.IDS) Not In (select IDS from [Delegati])));", 2)
If RC.EOF Then GoTo Start
Set TC = CurrentDb.OpenRecordset("Delegati", 2)

POP RC

Do While Not RC.EOF
    TC.AddNew
        TC!IDD = Nz(DMax("IDD", "Delegati"), 0) + 1
        TC!IDS = RC!IDS
        TC!Nume = RC!Nume
        TC!CNP = RC!CUI
    TC.Update
    RC.MoveNext
Loop

Start:
'======== preluare date in [GRUPE_L]
Set RC = CurrentDb.OpenRecordset("SELECT * FROM Grupe WHERE Nz(Grupa,'') <> '' ORDER BY Grupa", 2)
Set TC = CurrentDb.OpenRecordset("Grupe_L", 2)

GolesteTabel TC

POP RC

Do While Not RC.EOF
    TC.AddNew
        For i = 0 To RC.Fields.Count - 1
            TC.Fields("" & RC.Fields(i).Name & "").value = RC.Fields(i).value
        Next i
    TC.Update
    RC.MoveNext
Loop
Me.Grupe.Requery

'======== preluare date in [Platitori_L]
Set RC = CurrentDb.OpenRecordset("SELECT * FROM Platitori WHERE Nz(Nume,'') <> '' ORDER BY Nume", 2)
Set TC = CurrentDb.OpenRecordset("Platitori_L", 2)

GolesteTabel TC

POP RC

Do While Not RC.EOF
    TC.AddNew
        For i = 0 To RC.Fields.Count - 1
            TC.Fields("" & RC.Fields(i).Name & "").value = RC.Fields(i).value
        Next i
    TC.Update
    RC.MoveNext
Loop
Me.sub_Platitori.Requery

'======== preluare date in [Platitori_sub_L]
Set RC = CurrentDb.OpenRecordset("SELECT Platitori_sub.* FROM Platitori_sub INNER JOIN Platitori ON Platitori_sub.IDP=Platitori.IDP  WHERE Nz(Platitori_sub.Nume,'') <> '' ORDER BY Platitori_sub.Nume", 2)
Set TC = CurrentDb.OpenRecordset("Platitori_sub_L", 2)

GolesteTabel TC

POP RC

Do While Not RC.EOF
    TC.AddNew
        For i = 0 To RC.Fields.Count - 1
            TC.Fields("" & RC.Fields(i).Name & "").value = RC.Fields(i).value
        Next i
    TC.Update
    RC.MoveNext
Loop

'======== preluare date in [Delegati]
Set RC = CurrentDb.OpenRecordset("SELECT Delegati.* FROM Platitori INNER JOIN (Platitori_sub INNER JOIN Delegati ON Platitori_sub.IDS = Delegati.IDS) ON Platitori.IDP = Platitori_sub.IDP ORDER BY Delegati.IDD;", 2)
Set TC = CurrentDb.OpenRecordset("Delegati_l", 2)

GolesteTabel TC

POP RC

Do While Not RC.EOF
    TC.AddNew
        TC!IDD = RC!IDD
        TC!IDS = RC!IDS
        TC!Nume = RC!Nume
        TC!CNP = RC!CNP
        TC!DATEBI = RC!DATEBI
        TC!AUTO = RC!AUTO
    TC.Update
    RC.MoveNext
Loop

Me.Platitori_sub_L.Requery

Set RC = Nothing
Set TC = Nothing


End Sub

Private Sub Sav_Click()
On Error GoTo Eroare

Dim db As DAO.Database
Dim Wk As DAO.Workspace

Set Wk = DBEngine.Workspaces(0)
Set db = Wk.Databases(0)

Dim RC As Recordset, TC As Recordset, xc As DAO.Recordset

Wk.BeginTrans

Set RC = CurrentDb.OpenRecordset("SELECT * FROM Grupe_L WHERE Sters=True ORDER BY IDG", 2)
If RC.EOF Then GoTo FaraStergeGrupa

Set TC = db.OpenRecordset("SELECT * FROM Grupe WHERE IDG IN (SELECT IDG FROM Grupe_L WHERE Sters=True)", 2)
If Not TC.EOF Then
    TC.MoveLast
    Do While Not TC.BOF
        TC.Delete
        TC.MovePrevious
    Loop
End If

FaraStergeGrupa:

Set RC = db.OpenRecordset("SELECT * FROM Platitori_L WHERE Sters=True ORDER BY IDP", 2)
If RC.EOF Then GoTo FaraStergeCopil

Set TC = db.OpenRecordset("SELECT * FROM Platitori WHERE IDP IN (SELECT IDP FROM Platitori_L WHERE Sters=True)", 2)
If Not TC.EOF Then
    TC.MoveLast
    Do While Not TC.BOF
        TC.Delete
        TC.MovePrevious
    Loop
End If

FaraStergeCopil:
    
Set RC = db.OpenRecordset("SELECT * FROM Platitori_sub_L WHERE Sters=True ORDER BY IDS", 2)
If RC.EOF Then GoTo FaraStergePlatitor

Set TC = db.OpenRecordset("SELECT * FROM Platitori_sub WHERE IDP IN (SELECT IDP FROM Platitori_sub_L WHERE Sters=True)", 2)
If Not TC.EOF Then
    TC.MoveLast
    Do While Not TC.BOF
        TC.Delete
        TC.MovePrevious
    Loop
End If

FaraStergePlatitor:

Set RC = db.OpenRecordset("SELECT * FROM Grupe WHERE IDG IN (SELECT IDG FROM Grupe_L WHERE Editat=True) ORDER BY IDG", 2)
If RC.EOF Then GoTo FaraEditGrupa

POP RC

Do While Not RC.EOF
    RC.Edit
        RC!Grupa = DLookup("Grupa", "Grupe_L", "IDG=" & RC!IDG)
        RC!Educator = DLookup("educator", "Grupe_L", "IDG=" & RC!IDG)
    RC.Update
    RC.MoveNext
Loop

FaraEditGrupa:

Set RC = db.OpenRecordset("SELECT * FROM Platitori WHERE IDP IN (SELECT IDP FROM Platitori_L WHERE Editat=True) ORDER BY IDP", 2)
If RC.EOF Then GoTo FaraEditCopil

POP RC

Do While Not RC.EOF
    RC.Edit
        RC!IDG = DLookup("IDG", "Platitori_L", "IDP=" & RC!IDP)
        RC!Nume = DLookup("nume", "Platitori_L", "IDP=" & RC!IDP)
        RC!Adresa = DLookup("adresa", "Platitori_L", "IDP=" & RC!IDP)
        RC!CNP = DLookup("cnp", "Platitori_L", "IDP=" & RC!IDP)
        RC!Frate = DLookup("frate", "Platitori_L", "IDP=" & RC!IDP)
        RC!Plecat = DLookup("plecat", "Platitori_L", "IDP=" & RC!IDP)
        RC!IDF = DLookup("idf", "Platitori_L", "IDP=" & RC!IDP)
        RC!Avans = DLookup("avans", "Platitori_L", "IDP=" & RC!IDP)
        RC!SI = DLookup("si", "Platitori_L", "IDP=" & RC!IDP)
        RC!IDV = DLookup("idv", "Platitori_L", "IDP=" & RC!IDP)
        RC!DataIntrare = DLookup("DataIntrare", "Platitori_L", "IDP=" & RC!IDP)
        RC!DataIesire = DLookup("DataIesire", "Platitori_L", "IDP=" & RC!IDP)
    RC.Update
    
    Set xc = CurrentDb.OpenRecordset("SELECT * FROM Prezenta WHERE IDP = " & RC!IDP, 2)
    If POP(xc) Then
        Do While Not xc.EOF
            xc.Edit
                xc!IDG = DLookup("IDG", "Platitori_L", "IDP=" & RC!IDP)
            xc.Update
            xc.MoveNext
        Loop
    End If
    
    RC.MoveNext
Loop



FaraEditCopil:
    
Set RC = db.OpenRecordset("SELECT * FROM Platitori_sub WHERE IDS IN (SELECT IDS FROM Platitori_sub_L WHERE Editat=True) ORDER BY IDS", 2)
If RC.EOF Then GoTo FaraEditPlatitor

POP RC

Do While Not RC.EOF
    RC.Edit
        RC!Nume = DLookup("nume", "Platitori_sub_L", "IDS=" & RC!IDS)
        RC!Adresa = DLookup("adresa", "Platitori_sub_L", "IDS=" & RC!IDS)
        RC!CUI = DLookup("cui", "Platitori_sub_L", "IDS=" & RC!IDS)
        RC!J = DLookup("j", "Platitori_sub_L", "IDS=" & RC!IDS)
        RC!Cont = DLookup("cont", "Platitori_sub_L", "IDS=" & RC!IDS)
        RC!Banca = DLookup("Banca", "Platitori_sub_L", "IDS=" & RC!IDS)
        RC!EMail = DLookup("email", "Platitori_sub_L", "IDS=" & RC!IDS)
        RC!Activ = DLookup("activ", "Platitori_sub_L", "IDS=" & RC!IDS)
    RC.Update
    RC.MoveNext
Loop

FaraEditPlatitor:

Set RC = db.OpenRecordset("SELECT * FROM Delegati WHERE IDD IN (SELECT IDD FROM Delegati_L WHERE Editat=True) ORDER BY IDD", 2)
If RC.EOF Then GoTo FaraEditDelegat

POP RC

Do While Not RC.EOF
    RC.Edit
        RC!Nume = DLookup("Nume", "Delegati_L", "IDD=" & RC!IDD)
        RC!CNP = DLookup("CNP", "Delegati_L", "IDD=" & RC!IDD)
        RC!DATEBI = DLookup("DATEBI", "Delegati_L", "IDD=" & RC!IDD)
        RC!AUTO = DLookup("AUTO", "Delegati_L", "IDD=" & RC!IDD)
    RC.Update
    RC.MoveNext
Loop

FaraEditDelegat:

Call GolesteTabel(, "Grupe_L", "Editat = True AND Sters = True")
Call GolesteTabel(, "Platitori_L", "Editat = True AND Sters = True")
Call GolesteTabel(, "Platitori_sub_L", "Editat = True AND Sters = True")
Call GolesteTabel(, "Delegati_L", "Editat = True")

Set RC = db.OpenRecordset("Grupe", 2)
Set TC = db.OpenRecordset("SELECT * FROM Grupe_L WHERE IDG NOT IN (SELECT IDG FROM Grupe) AND Sters=False AND Editat=False", 2)

If TC.EOF Then GoTo FaraGrupeNoi

POP TC

Do While Not TC.EOF
    RC.AddNew
        RC!IDG = TC!IDG
        RC!Grupa = TC!Grupa
        RC!Educator = TC!Educator
    RC.Update
    TC.MoveNext
Loop

FaraGrupeNoi:

Set RC = db.OpenRecordset("Platitori", 2)
Set TC = db.OpenRecordset("SELECT * FROM Platitori_L WHERE IDP NOT IN (SELECT IDP FROM Platitori) AND Sters=False AND Editat=False", 2)

If TC.EOF Then GoTo FaraCopiiNoi

POP TC

Do While Not TC.EOF
    RC.AddNew
        RC!IDP = TC!IDP
        RC!IDG = TC!IDG
        RC!Nume = TC!Nume
        RC!Adresa = TC!Adresa
        RC!CNP = TC!CNP
        RC!Frate = TC!Frate
        RC!Plecat = TC!Plecat
        RC!Avans = TC!Avans
        RC!SI = TC!SI
        RC!IDV = TC!IDV
    RC.Update
    TC.MoveNext
Loop

FaraCopiiNoi:

Set RC = db.OpenRecordset("Platitori_sub", 2)
Set TC = db.OpenRecordset("SELECT * FROM Platitori_sub_L WHERE Nou=True", 2)

If TC.EOF Then GoTo FaraPlatitoriNoi

POP TC

Do While Not TC.EOF
    RC.AddNew
        RC!IDP = TC!IDP
        RC!IDS = TC!IDS
        RC!Nume = TC!Nume
        RC!Adresa = TC!Adresa
        RC!CUI = TC!CUI
        RC!J = TC!J
        RC!Cont = TC!Cont
        RC!Banca = TC!Banca
        RC!EMail = TC!EMail
        RC!Activ = TC!Activ
    RC.Update
    TC.MoveNext
Loop

FaraPlatitoriNoi:

Set RC = db.OpenRecordset("Delegati", 2)
Set TC = db.OpenRecordset("SELECT * FROM Delegati_L WHERE Nou=True", 2)

If TC.EOF Then GoTo FaraDelegatiNoi

POP TC

Do While Not TC.EOF
    RC.AddNew
        RC!IDD = TC!IDD
        RC!IDS = TC!IDS
        RC!Nume = TC!Nume
        RC!CNP = TC!CNP
        RC!DATEBI = TC!DATEBI
        RC!AUTO = TC!AUTO
        RC!Activ = TC!Activ
    RC.Update
    TC.MoveNext
Loop

FaraDelegatiNoi:

Wk.CommitTrans

MsgBox "Salvare efectuata cu succes!", vbOKOnly + vbInformation

Set RC = Nothing
Set TC = Nothing
Set db = Nothing
Set Wk = Nothing

Call Form_Open(0)
Exit Sub

Eroare:
    Wk.Rollback
    MsgBox "Eroare:" & vbCrLf & Err.Description & vbCrLf & Err.Number
End Sub
