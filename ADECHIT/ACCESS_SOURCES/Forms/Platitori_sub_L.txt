Private Sub Activ_Click()
Dim RC As Recordset

Set RC = CurrentDb.OpenRecordset("SELECT * FROM Platitori_sub_L WHERE IDS <> " & Me.IDS & " AND IDP = " & Me.IDP, 2)
POP RC

If Me.Dirty Then Me.Dirty = False

If Not Activ Then
    Activ = Not Activ
    MsgBox "Nu se poate dezactiva Platitorul!" & vbCrLf & "Bifeaza altul si devine acela activ!", vbInformation + vbOKOnly
Else
    Do While Not RC.EOF
        RC.Edit
            RC!Activ = False
        RC.Update
        RC.MoveNext
    Loop
End If

Set RC = Nothing

If Me.Dirty Then Me.Dirty = False

End Sub

Private Sub Form_Delete(Cancel As Integer)
If Activ = True Then
    MsgBox "Nu poti sterge inregistrarea activa!", vbOKOnly
    Cancel = True
End If

If MsgBox("Doresti stergerea platitorului selectat?", vbYesNo + vbQuestion) = vbYes Then
    Cancel = True
    Select Case StergeRand("Platitori_sub_L", "ids = " & Me.IDS)
        Case -1
        Case 1
            MsgBox "Eroare la stergerea Platitorului!", vbOKOnly + vbCritical
        Case 0
            MsgBox "Imposibil sa ajungi aici!", vbOKOnly
    End Select
Else
    Cancel = True
End If

Me.TimerInterval = 100
End Sub
