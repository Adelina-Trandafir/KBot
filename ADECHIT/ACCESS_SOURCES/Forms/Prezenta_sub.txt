Private Sub Sav_Click()
If MsgBox("Doresti salvarea datelor pentru luna " & Forms!prezenta!LunaT & "/" & Forms!prezenta!Anul & " si grupa " & Forms!prezenta!Grupe!Grupa & "?", vbYesNo + vbQuestion) = vbNo Then Exit Sub

On Error GoTo Eroare

Dim db As DAO.Database
Dim Wk As DAO.Workspace
Dim i As Integer

Set Wk = DBEngine.Workspaces(0)
Set db = Wk.Databases(0)

Dim RC As Recordset, TC As Recordset

Wk.BeginTrans

Set RC = db.OpenRecordset("SELECT * FROM LunaD_L WHERE IDL NOT IN (SELECT IDL FROM LunaD) AND IDL IN (SELECT IDL FROM Prezenta_L)", 2)
POP RC

If RC.EOF Then 'nu s-au adaugat luni noi
    GoTo FaraLunaNoua
End If

Set TC = db.OpenRecordset("LunaD", 2)
Do While Not RC.EOF
    TC.AddNew
        TC!IDL = RC!IDL
        TC!IDV = RC!IDV
        TC!Luna = RC!Luna
        TC!Anul = RC!Anul
        TC!LunaT = RC!LunaT
        TC!LA = RC!LA
    TC.Update
    RC.MoveNext
Loop

FaraLunaNoua:

Call GolesteTabel(, "Prezenta", "IDZ IN (SELECT IDZ FROM Prezenta_L)") 'elimin din baza copii in lucru
Call GolesteTabel(, "Prezenta_L", "Sters=True") 'elimin copii stersi din temporar

Set RC = db.OpenRecordset("SELECT * FROM Prezenta_L", 2)
If RC.EOF Then 'ciudat
    GoTo FaraPrezenta
End If

Set TC = db.OpenRecordset("Prezenta", 2)
Do While Not RC.EOF
    TC.AddNew
        TC!IDZ = RC!IDZ
        TC!IDP = RC!IDP
        TC!IDL = RC!IDL
        TC!IDV = RC!IDV
        TC!IDG = RC!IDG
        TC!ZileLuna = RC!ZileLuna
        TC!ZilePrezenta = RC!ZilePrezenta
        TC!ZileAbsenta = RC!ZileAbsenta
        TC!MZCPrezenta = RC!MZCPrezenta
        TC!MZCAbsenta = RC!MZCAbsenta
        TC!ValoareContract = RC!ValoareContract
        TC!ValoareMancare = RC!ValoareMancare
        TC!ValoareTotala = RC!ValoareTotala
        TC!ReducereFrate = RC!ReducereFrate
        TC!ReducereBonus = RC!ReducereBonus
        TC!SI = RC!SI
        TC!Detalii = RC!Detalii
        TC!Avans = RC!Avans
    TC.Update
    RC.MoveNext
Loop

Set RC = db.OpenRecordset("SELECT * FROM Prezenta_sub_l", 2)
Set TC = db.OpenRecordset("Prezenta_sub", 2)

Do While Not RC.EOF
    TC.AddNew
        TC!idx = RC!idx
        TC!IDZ = RC!IDZ
        TC!IDL = RC!IDL
        TC!DI = RC!DI
        TC!DSF = RC!DSF
        TC!nrzile = RC!nrzile
        TC!absent = RC!absent
    TC.Update
    RC.MoveNext
Loop

FaraPrezenta:

Wk.CommitTrans

MsgBox "Salvare efectuata cu succes!", vbOKOnly + vbInformation

Set RC = Forms!prezenta!Luni.Form.Recordset
    RC.FindFirst "IDL = " & Forms!prezenta!IDL

Set RC = Forms!prezenta!Grupe.Form.Recordset
    RC.FindFirst "IDG = " & Forms!prezenta!IDG
    
Forms!prezenta!Luni.Enabled = True
Forms!prezenta!Luni.Form.AllowAdditions = True

Forms!prezenta!Grupe.Enabled = True

Forms!prezenta!Prezenta_sub.Enabled = False
IESIRE:
Set RC = Nothing
Set TC = Nothing
Set db = Nothing
Set Wk = Nothing

Exit Sub

Eroare:
    Wk.Rollback
    MsgBox "Eroare salvare luna: " & vbCrLf & Err.Number & vbCrLf & Err.Description
    Resume IESIRE
End Sub
