Option Compare Database

Private OA As String

Private Sub Form_Load()
OA = Nz(Me.OpenArgs, "")
Select Case OA
    Case ""
        Me.Caption = Split(Me.Tag, "|")(0)
        Me.C0.RowSource = "SELECT Grupe.IDG, Grupe.Grupa, Grupe.Educator FROM Grupe WHERE Grupe.IDG Not Like " & [Forms]![prezenta]![IDG] & " ORDER BY Grupe.Grupa; "
    Case "TRG"
        Me.Caption = Split(Me.Tag, "|")(1)
        Me.C0.RowSource = "SELECT Grupe.IDG, Grupe.Grupa, Grupe.Educator FROM Grupe WHERE Grupe.IDG Not Like " & [Forms]![platitori2016]![Grupe]![IDG] & " ORDER BY Grupe.Grupa; "
End Select

End Sub

Private Sub IESIRE_Click()
If OA = "TRG" Then
    Forms!platitori2016!sub_Platitori.Form.Requery
    CurrentDb.Execute "UPDATE Platitori_L SET S=False"
End If

DoCmd.Close acForm, Me.Name
End Sub

Private Sub Sav_Click()
Dim MaxIDZ As Long

If Nz(C0, "") = "" Then
    MsgBox "Nu ai ales nicio grupa!", vbOKOnly + vbInformation
    DoCmd.Close acForm, Me.Name
    Exit Sub
End If

If OA = "TRG" Then GoTo TransferGrupa

If MsgBox("Doresti sa transferi copilul " & Forms!prezenta!Prezenta_sub!IDP_p & " din grupa (" & DLookup("Grupa", "Grupe", "IDG=" & Forms!prezenta!IDG) & ") la grupa {" & C0.Column(1) & "}?", vbYesNo + vbQuestion) = vbNo Then Exit Sub

Dim RC As Recordset

Set RC = CurrentDb.OpenRecordset("SELECT * FROM Platitori WHERE IDP=" & Forms!prezenta!IDP, 2)
RC.Edit
    RC!IDG = C0
RC.Update

Set RC = CurrentDb.OpenRecordset("SELECT * FROM Prezenta WHERE IDZ=" & Forms!prezenta!IDZ, 2)
RC.Edit
    RC!IDG = C0
RC.Update

Set RC = CurrentDb.OpenRecordset("MutaCopil", 2)
RC.AddNew
    RC!idm = Nz(DMax("IDM", "MutaCopil"), 0) + 1
    RC!IDL = Forms!prezenta!IDL
    RC!IDP = Forms!prezenta!IDP
    RC!IDZ = Forms!prezenta!Prezenta_sub!IDZ
    RC!idg_n = C0
    RC!idg_v = Forms!prezenta!IDG
    RC!Data = Format(Now(), "Short Date")
RC.Update

Forms!prezenta!Prezenta_sub!IDP_p.Requery
Forms!prezenta!Prezenta_sub.Requery
DoCmd.Close acForm, Me.Name

Exit Sub

TransferGrupa:
CurrentDb.Execute "UPDATE Platitori SET IDG = " & Me!C0 & " WHERE IDP IN (SELECT IDP FROM Platitori_L WHERE S=True)"
DoCmd.Close acForm, Me.Name, acSaveNo
DoCmd.Close acForm, "Platitori2016", acSaveNo
DoCmd.OpenForm "Platitori2016", acNormal

End Sub
