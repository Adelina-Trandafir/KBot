Private Sub Anuleaza_Click()
If Listata Then
    If MsgBox("Atentie! Aceasta factura a fost listata cel putin o data!" & vbCrLf & "Esti sigura ca doresti sa o stergi?", vbYesNo + vbQuestion) = vbNo Then Exit Sub
Else
    If MsgBox("Doresti anularea facturii curente?", vbYesNo + vbQuestion) = vbNo Then Exit Sub
End If

Anulata = True

Dim RC As Recordset
Set RC = CurrentDb.OpenRecordset("SELECT * FROM Facturi WHERE IDPL=" & IDPL, 2)
    RC.Edit
        RC!Anulata = True
    RC.Update
Set RC = Nothing

End Sub

IDS = Beneficiar
Me.Requery

End Sub


Private Sub Data_Change()

If CDate(Nz(Data.Text, #12/31/2100#)) < Forms!prezenta!Plati!Data Then
    MsgBox "Data facturii nu poate fi mai mica decat data platii!", vbOKOnly + vbCritical
    Data.Undo
End If

Me.Recalc

End Sub

Private Sub Form_Load()
IDPL = Forms!prezenta!Plati!IDPL
If IDPL = 0 Then
    MsgBox "Cum ai ajuns aici?", vbOKOnly + vbQuestion
    Exit Sub
End If

Dim RC As Recordset
Set RC = CurrentDb.OpenRecordset("SELECT * FROM Facturi WHERE IDPL=" & IDPL, 2)
If RC.EOF Then
    IDF = Nz(DMax("IDF", "Facturi"), 0) + 1
    IDZ = DLookup("IDZ", "Plati", "IDPL = " & IDPL)
    IDS = DLookup("IDS", "Plati", "IDPL = " & IDPL)
    IDP = DLookup("IDP", "Plati", "IDPL = " & IDPL)
    IDL = DLookup("IDL", "Plati", "IDPL = " & IDPL)
    IDD = DLookup("IDD", "Delegati", "IDS=" & IDS)
    
    Call CalculVariabile(IDP, IDL, IDZ, IDPL)
    
    Serie = DLookup("F_SERIE", "Unitati")
    Numar = IIf(Nz(DMax("NUMAR", "Facturi"), 0) > Nz(DMax("NUMAR", "Facturi"), 0), Nz(DMax("NUMAR", "Facturi"), 0) + 1, Nz(DMax("NUMAR", "Facturi"), 0) + 1)
    Beneficiar.Requery
    Beneficiar = IDS
    Continut = fMail(DLookup("f_COntinut", "Unitati"))
    Rand1 = fMail(Nz(DLookup("f_R1", "Unitati"), ""))
    Rand2 = fMail(Nz(DLookup("F_R2", "Unitati"), ""))
    UM = "Buc."
    Cant = "1"
    PU = DLookup("Plata", "plati", "idpl=" & IDPL)
    Valoare = PU
    TOT2 = PU
    Listata = False
    Anulata = False
    Nou = True
    Anuleaza.Enabled = False
    Listeaza.Enabled = False
    EMail.Enabled = False
    Data = Forms!prezenta!Plati!Data
Else
    IDF = RC!IDF
    IDZ = DLookup("IDZ", "Plati", "IDPL = " & IDPL)
    IDS = RC!IDS
    IDP = DLookup("IDP", "Plati", "IDPL = " & IDPL)
    IDL = DLookup("IDL", "Plati", "IDPL = " & IDPL)
    IDD = RC!IDD
        
    Serie = RC!Serie
    Numar = RC!Numar
    Beneficiar.Requery
    Beneficiar = IDS
    Continut = RC!Continut
    Rand1 = RC!Rand1
    Rand2 = RC!Rand2
    UM = RC!UM
    Cant = RC!Cant
    PU = RC!PU
    Valoare = RC!Valoare
    TOT2 = Valoare
    Data = RC!Data
    Anulata = RC!Anulata
    Listata = RC!Listata
    Nou = False
    Anuleaza.Enabled = True
    Listeaza.Enabled = True
    EMail.Enabled = True
End If

If Anulata Then
    Label7.ForeColor = RGB(255, 0, 0)
    Label7.ControlTipText = "#FACTURA ANULATA#"
    Label8.ForeColor = RGB(255, 0, 0)
    Data.Enabled = False
    Beneficiar.Enabled = False
    Continut.Enabled = False
    sav.Enabled = False
    EMail.Enabled = False
Else
    Label7.ForeColor = RGB(0, 0, 0)
    Label8.ForeColor = RGB(0, 0, 0)
    Data.Enabled = True
    Beneficiar.Enabled = True
    Continut.Enabled = True
    sav.Enabled = True
End If

End Sub

Dim RC As Recordset
Set RC = CurrentDb.OpenRecordset("SELECT * FROM Facturi WHERE IDPL=" & IDPL, 2)
    RC.Edit
        RC!Listata = True
    RC.Update
Set RC = Nothing


DoCmd.OpenReport "rpt_Factura", acViewPreview
End Sub

Private Sub Sav_Click()
If Nz(Data, "") = "" Then
    MsgBox "Nu ai introdus data facutrii! Nu se poate salva!", vbOKOnly + vbCritical
    Exit Sub
End If

If Listata Then If MsgBox("Factura a fost listata!" & vbCrLf & "Nu mai poate fi modificata!", vbOKOnly + vbCritical) <> vbYes Then Exit Sub
    
If MsgBox("Doresti salvarea facturii curente?", vbYesNo + vbQuestion) = vbNo Then Exit Sub

Dim RC As Recordset
If Nou Then
    Set RC = CurrentDb.OpenRecordset("Facturi", 2)
        RC.AddNew
Else
    Set RC = CurrentDb.OpenRecordset("SELECT * FROM Facturi WHERE IDF=" & IDF, 2)
        RC.Edit
End If

RC!IDF = IDF
RC!IDPL = IDPL
RC!IDD = IDD
RC!IDS = IDS
RC!Serie = Serie
RC!Numar = Numar
RC!Data = Data
RC!Continut = Continut
RC!UM = UM
RC!Cant = Cant
RC!PU = PU
RC!Valoare = Valoare
RC!Anulata = Anulata
RC!Listata = Listata
RC!Rand1 = Rand1
RC!Rand2 = Rand2

RC.Update

MsgBox "Salvare efectuata cu succes!", vbOKOnly + vbInformation

FOC.SetFocus
Anuleaza.Enabled = True
Listeaza.Enabled = True
sav.Enabled = False
EMail.Enabled = True

End Sub
