Private Sub Activ_Click()
If Me.NewRecord Then
    MsgBox "Nu poti activa randul nou!" & vbCrLf & "Acesta se activeaza singur!", vbOKOnly + vbInformation
    Activ = Not Activ
    GolesteTabel , "ValoriTaxe", "IDV=" & Me.IDV
    Me.Requery
    Exit Sub
End If

If Not Activ Then
    MsgBox "Nu poti dezactiva randul curent!" & vbCrLf & "Adauga alt rand, si activeaza-l pe acela!", vbOKOnly + vbInformation
    Activ = Not Activ
    Exit Sub
Else
    Dim RC As Recordset
    
    Set RC = CurrentDb.OpenRecordset("SELECT * FROM ValoriTaxe WHERE IDV <> " & Me.IDV, 2)
    
    Do While Not RC.EOF
        RC.Edit
            RC!Activ = False
        RC.Update
        RC.MoveNext
    Loop
    
    Set RC = Nothing
End If

End Sub

If Nz(TaxaLunara, "") = "" Then e = True

If Nz(TaxaMancare, "") = "" Then e = True

If e = True Then
    MsgBox "Nu ai completat toate informatiile!", vbOKOnly + vbCritical
    GolesteTabel , "ValoriTaxe", "IDV = " & Me.IDV
    Me.Requery
    Exit Sub
End If

Me.Activ = True

Dim RC As Recordset

Set RC = CurrentDb.OpenRecordset("SELECT * FROM ValoriTaxe WHERE IDV <> " & Me.IDV, 2)

Do While Not RC.EOF
    RC.Edit
        RC!Activ = False
    RC.Update
    RC.MoveNext
Loop

Set RC = Nothing

End Sub
