    MsgBox "Plata selectata are factura activa atasata!" & vbCrLf & "Aceasta nu poate fi stearsa!", vbOKOnly + vbCritical
    Exit Sub
End If

Select Case TIP
    Case "1"
        If DCount("IDB", "BonuriF", "IDPL=" & IDPL & " AND Corect=True") <> 0 Then
            If MsgBox("Plata selectata are bon fiscal atasat si listat corect!" & vbCrLf & "Prin steregerea randului curent se va sterge si acest bon, care poate creea diferente intre [Borderu] din program si [Raport Z] din casa de marcat!" & vbCrLf & "Continui?", vbYesNo + vbQuestion) = vbNo Then Exit Sub
        End If
    Case "2"
        If CBool(DLookup("Anulata", "Chitante", "IDPL=" & IDPL)) = True Then
            MsgBox "Chitanta selectata este anulata!", vbOKOnly + vbInformation
            Exit Sub
        End If
        
        If MsgBox("Chitantele NU pot fi sterse, decat anulate!" & vbCrLf & "Doresti anularea chitantei " & DLookup("Serie", "Chitante", "IDPL=" & IDPL) & "/" & DLookup("Numar", "Chitante", "IDPL=" & IDPL) & "?" & vbCrLf & "ATENTIE!         Me!Anulata = True
        Me.Recalc
        
        Set RC = CurrentDb.OpenRecordset("SELECT * FROM Chitante WHERE IDPL = " & IDPL, 2)
            RC.Edit
                RC!Anulata = True
            RC.Update
            
        Set RC = Nothing
        
        MsgBox "Chitanta a fost anulata!", vbOKOnly + vbInformation
        
        Forms!prezenta!Facturi.Requery
        
        Exit Sub
