Private Sub Sav_Click()
Dim MaxIDZ As Long

If Nz(C0, "") = "" Then
    MsgBox "Nu ai ales niciun copil!", vbOKOnly + vbInformation
    DoCmd.Close acForm, Me.Name
    Exit Sub
End If

Dim RC As Recordset, TC As Recordset, Qc As DAO.QueryDef

Set RC = CurrentDb.OpenRecordset("SELECT * FROM Prezenta_L", 2)

RC.AddNew
    MaxIDZ = Nz(DMax("IDZ", "Prezenta_l"), 0)
    If MaxIDZ < Nz(DMax("IDZ", "Prezenta"), 0) Then MaxIDZ = Nz(DMax("IDZ", "Prezenta"), 0)
    RC!IDZ = MaxIDZ + 1
    RC!IDP = C0
    RC!IDL = Forms!prezenta!IDL
    RC!IDV = DLookup("IDV", "ValoriTaxe", "Activ=True")
    RC!IDG = Forms!prezenta!IDG
RC.Update

GolesteTabel , "tmpSitDebPrezenta"

Set Qc = CurrentDb.QueryDefs("SItLunara")
    Qc.Parameters(0) = [Forms]![prezenta]![IDL]

Set RC = Qc.OpenRecordset

Set TC = CurrentDb.OpenRecordset("tmpSitDebPrezenta", 2)

If POP(RC) Then
    Do While Not RC.EOF
        TC.AddNew
            TC!IDP = RC!IDP
            TC!SID = RC!SID
            TC!SIC = Abs(RC!SIC)
            TC!Retur = RC!Retur
            TC!Platita = RC!Platita
        TC.Update
        
        RC.MoveNext
    Loop
End If

Forms!prezenta!Prezenta_sub!IDP_p.Requery
Forms!prezenta!Prezenta_sub.Requery
DoCmd.Close acForm, Me.Name

End Sub
