Private Sub Sav_Click()
Dim Rest As Double, Achit As Double, ValLuna As Double, ValAnti As Double

If Nz(Continut, "") = "" Then
    MsgBox "Completeaza toate campurile!", vbOKOnly + vbCritical
    Exit Sub
End If

If Nz(DCount("IDF", "facturi", "IDPL=" & IDPL), 0) <> 0 Then
    MsgBox "Plata curenta are factura salvata!" & vbCrLf & "Nu se mai pot modifica datele curente!", vbOKOnly + vbCritical
    Exit Sub
End If

If MsgBox("Doresti salvarea datelor?", vbYesNo + vbQuestion) = vbNo Then Exit Sub

Dim RC As Recordset
Set RC = CurrentDb.OpenRecordset("SELECT * FROM Chitante WHERE IDPL=" & IDPL, 2)

If RC.EOF Then
    RC.AddNew
        RC!IDC = IDC
        RC!IDPL = IDPL
        RC!Data = DataCH
        RC!Numar = NrCH
        RC!Valoare = Valoare
        RC!Explicatie = Continut
        RC!Anulata = Anulata
        RC!Serie = Nz(DLookup("VL", "CFGs", "F='CH' AND CFG='SERIE'"), "CH")
    RC.Update
    
    ActualizeazaCamp "CFGs", "VL", NrCH, "F='CH' AND CFG='NUMAR'"
    
    Achit = Nz(DSum("Plata", "Plati", "IDP = " & Forms!prezenta!IDP & " AND IDZ = " & Forms!prezenta!IDZ & " AND Anulata = False"), 0)
    
    ValLuna = Nz(DLookup("ValoareContract", "Prezenta", "IDZ=" & Forms!prezenta!IDZ), 0)
    ValAnti = Nz(DLookup("Restanta", "Prezenta", "IDZ=" & Forms!prezenta!IDZ & " AND Restanta < 0"), 0)
    
    Rest = Nz(DLookup("Restanta", "Prezenta", "IDZ=" & Forms!prezenta!IDZ & " AND Restanta > 0"), 0) - Nz(DSum("Restanta", "Plati", "IDP = " & Forms!prezenta!IDP & " AND IDZ = " & Forms!prezenta!IDZ & " AND Anulata = False"), 0)
    
    Set RC = CurrentDb.OpenRecordset("SELECT * FROM Plati WHERE IDPL = " & IDPL, 2)
        RC.Edit
            If Rest > 0 Then
                If Achit <= Rest Then
                    RC!Restanta = RC!Plata
                Else
                    If Achit <= RC!Valoare Then
                        RC!Restanta = Rest
                    Else
                        RC!Restanta = Rest
                        RC!Anticipat = RC!Plata - Rest - ValLuna
                    End If
                End If
            Else
                If Achit > RC!Valoare Then
                    RC!Anticipat = Achit - ValLuna - ValAnti
                End If
            End If
                
            RC!Valid = True
        RC.Update
    
    cSAV = True
    
Else
    If Anulata Then
        If MsgBox("Odata anulata o chitanta ACEASTA NU MAI POATE FI RESTAURATA NICIODATA!!!!" & vbCrLf & "Doresti anularea chitantei curente?", vbYesNo + vbQuestion) = vbNo Then
            Set RC = Nothing
            Exit Sub
        End If
        
        RC.Edit
            RC!Anulata = Anulata
        RC.Update
        
        MsgBox "Chitanta a fost anulata!", vbOKOnly + vbInformation
        
        Set RC = Nothing
        
        Exit Sub
        
    Else
        MsgBox "O chitanta salvata nu mai poate fi modificata, DECAT anulata!", vbOKOnly + vbInformation
        Exit Sub
    End If
End If

MsgBox "Salvare efectuata cu succes!", vbOKOnly + vbInformation

Set RC = Nothing

Forms!prezenta!Plati!Murdar = False

End Sub
