Private Sub Del_Click()
If IsLoaded("Platitori2016") Then
    If Me.NewRecord Then
        MsgBox "Nu poti sterge randul pentru introducerea datelor!", vbOKOnly + vbInformation
        Exit Sub
    End If
    
    If MsgBox("Doresti stergerea grupei selectate?" & vbCrLf & "Aceasta presupune stergerea TUTUROR copiilor din aceasta grupa!!!", vbYesNo + vbCritical) = vbNo Then Exit Sub
    
    Select Case StergeRand("Grupe_L", "IDG = " & Me.IDG)
        Case -1
            Me.Requery
            Parent!sub_Platitori.Requery
            Parent!Platitori_sub_L.Requery
        Case 1
            MsgBox "Eroare la stergerea Grupei!", vbOKOnly + vbCritical
        Case 0
            MsgBox "Imposibil sa ajungi aici!", vbOKOnly
    End Select
    
    Exit Sub
End If

End Sub

Private Sub Form_BeforeUpdate(Cancel As Integer)

End Sub


Private Sub Form_Current()
If SeDeschide Then Exit Sub

Nou = False
CL = Me.IDG
If IsLoaded("Platitori2016") Then
    Forms!platitori2016!IDG = Me.IDG
    Forms!platitori2016!sub_Platitori.Requery
    If Parent!S = True Then
        Parent!S = False
        Parent!bTransferGrupa.Caption = Split(Parent!bTransferGrupa.Tag, "|")(0)
        Parent!sub_Platitori!S.visible = False
        CurrentDb.Execute "UPDATE Platitori_L SET S=False"
    End If
    Exit Sub
End If
        
End Sub
