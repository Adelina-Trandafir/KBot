Option Explicit

Private Type POINTAPI
  x As Long
  y As Long
End Type

Private Type MOUSEHOOKSTRUCT
  PT As POINTAPI
  hWnd As LongPtr
  wHitTestCode As Long
  dwExtraInfo As LongPtr
End Type

' Based on code from Peter Thornton here:
' http://social.msdn.microsoft.com/Forums/en-US/7d584120-a929-4e7c-9ec2-9998ac639bea/mouse-scroll-in-userform-listbox-in-excel-2010?forum=isvvba
Private Declare PtrSafe Function CallNextHookEx Lib "user32" (ByVal hHook As LongPtr, ByVal ncode As Long, ByVal wParam As LongPtr, lParam As Any) As LongPtr
Private Declare PtrSafe Function GetCursorPos Lib "user32" (lpPoint As POINTAPI) As Long
Private Declare PtrSafe Function GetWindowLong Lib "user32" Alias "GetWindowLongA" (ByVal hWnd As LongPtr, ByVal nIndex As Long) As Long
Private Declare PtrSafe Function SetWindowsHookEx Lib "user32" Alias "SetWindowsHookExA" (ByVal idHook As Long, ByVal lpfn As LongPtr, ByVal hmod As LongPtr, ByVal dwThreadId As Long) As LongPtr
Private Declare PtrSafe Function WindowFromPoint Lib "user32" (ByVal xPoint As Long, ByVal yPoint As Long) As LongPtr
Private Declare PtrSafe Function UnhookWindowsHookEx Lib "user32" (ByVal hhk As LongPtr) As Long

Private Const WH_MOUSE_LL          As Long = 14
Private Const WM_MOUSEWHEEL        As Long = &H20A
Private Const HC_ACTION            As Long = 0
Private Const GWL_HINSTANCE        As Long = (-6)

Private Const cSCROLLCHANGE        As Long = 20

Public mLngMouseHook              As LongPtr
Public mFrameHwnd                 As LongPtr
Public mbHook                     As Boolean
Public mCtl                       As MSForms.Control
Dim n                              As Long


Sub HookFrameScroll(frm As Object, ctl As MSForms.Control)
Const C_PROC_NAME = "mdl_MouseScroll_Treeview/HookFrameScroll"
'ADD_TRACE C_PROC_NAME

On Error GoTo Eroare

Dim lngAppInst As Long
Dim hwndUnderCursor As LongPtr
Dim tPT As POINTAPI

GetCursorPos tPT

hwndUnderCursor = WindowFromPoint(tPT.x, tPT.y)

'>> uncomment below if you want to give the treeview focus once the user scrolls it
If Not frm.ActiveControl Is ctl Then ctl.SetFocus

If mFrameHwnd <> hwndUnderCursor Then
    UnhookFrameScroll
    
    Set mCtl = ctl
    
    mFrameHwnd = hwndUnderCursor
    
    lngAppInst = GetWindowLong(mFrameHwnd, GWL_HINSTANCE)
    
    If Not mbHook Then
        mLngMouseHook = SetWindowsHookEx(WH_MOUSE_LL, AddressOf FrameMouseProc, lngAppInst, 0)
        
        mbHook = mLngMouseHook <> 0
    End If
End If

IESIRE:
'REM_TRACE C_PROC_NAME

Exit Sub

Eroare:
    UnhookFrameScroll
    
    DebugPrint "EROARE: " & Err.Description, C_PROC_NAME
    Resume IESIRE
    
End Sub

Public Sub UnhookFrameScroll()
Const C_PROC_NAME = "mdl_MouseScroll_Treeview/UnhookFrameScroll"
'ADD_TRACE C_PROC_NAME

On Error GoTo Eroare

If mbHook Then
    Set mCtl = Nothing
    UnhookWindowsHookEx mLngMouseHook
    mLngMouseHook = 0
    mFrameHwnd = 0
    mbHook = False
End If

IESIRE:
'REM_TRACE C_PROC_NAME

Exit Sub

Eroare:
    UnhookFrameScroll
    
    DebugPrint "EROARE: " & Err.Description, C_PROC_NAME
    Resume IESIRE
End Sub


Private Function FrameMouseProc(ByVal ncode As Long, ByVal wParam As Long, ByRef lParam As MOUSEHOOKSTRUCT) As LongPtr
Const C_PROC_NAME = "mdl_MouseScroll_Treeview/FrameMouseProc"

On Error GoTo Eroare
    
If (ncode = HC_ACTION) Then
    If WindowFromPoint(lParam.PT.x, lParam.PT.y) = mFrameHwnd Then
        If wParam = WM_MOUSEWHEEL Then
            FrameMouseProc = True
            If lParam.hWnd > 0 Then
                mCtl.ScrollTop = iMax(0, mCtl.ScrollTop - cSCROLLCHANGE)
            Else
                mCtl.ScrollTop = iMin(mCtl.ScrollHeight - mCtl.InsideHeight, mCtl.ScrollTop + cSCROLLCHANGE)
            End If
        End If
    End If
End If


IESIRE:
FrameMouseProc = CallNextHookEx(mLngMouseHook, ncode, wParam, ByVal lParam)

Exit Function

Eroare:
    UnhookFrameScroll
    DebugPrint "EROARE: " & Err.Description, C_PROC_NAME
    Resume IESIRE
End Function

Public Function iMax(ParamArray p()) As Variant
  ' Idea from Trevor Best in Usenet MessageID rib5dv45ko62adf2v0d1cot4kiu5t8mbdp@4ax.com
    Dim i As Long
    Dim v As Variant

    v = p(LBound(p))
    For i = LBound(p) + 1 To UBound(p)
      If v < p(i) Then
         v = p(i)
      End If
    Next
    iMax = v
End Function
