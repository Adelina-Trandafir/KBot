Private Sub DeletePopUpMenu(Mname As String)
On Error GoTo Eroare

Dim obj As Object

For Each obj In Application.CommandBars
    If obj.Name = Mname Then
        Application.CommandBars(Mname).Delete
    End If
Next

Exit Sub

Eroare:
    MsgBox Err.Description & vbCrLf & "Modules/Popup/DeletePopupMenu()"
End Sub

Sub fPP_Prezenta()
Popup_Var1 = CommandBars.ActionControl.Parameter

Dim frm As Form, ctl As Control
Dim TipProgram As String

TipProgram = Nz(DLookup("VL", "CFGs", "F='XX' AND CFG='Platitor'"), "Buget")

If IsLoaded("Prezenta") Then
    Set frm = Forms!prezenta.Form
Else
    Set frm = Nothing
    Set ctl = Nothing
    Exit Sub
End If

If frm!Inchisa Then
    Select Case Popup_Var1
        Case 1, 2
            MsgBox "Luna selectata este inchisa! Nu se mai pot face modificari!", vbOKOnly + vbCritical
            Set frm = Nothing
            Set ctl = Nothing
            Exit Sub
            
        Case 3, 5
            If Nz(DSum("zilePrezenta", "Prezenta", "IDG=" & Forms!prezenta!IDG & " AND IDL = " & Forms!prezenta!IDL + 1), 0) <> 0 Then
                MsgBox "A fost introdusa prezenta pentru grupa " & DLookup("Grupa", "Grupe", "IDG=" & Forms!prezenta!IDG) & "/" & DLookup("LunaT & '/' & Anul", "LunaD", "IDL=" & Forms!prezenta!IDL + 1) & "!" & vbCrLf & "Nu se mai pot adauga / modifica plati!", vbOKOnly + vbCritical
                Set frm = Nothing
                Set ctl = Nothing
                Exit Sub
            End If
    End Select
End If

If Forms!prezenta!IDL <> Forms!prezenta!IDL Then
    Call IncarcaPrezenta
Else
    If Forms!prezenta!Prezenta_sub.Form.Recordset.RecordCount = 0 Then Call IncarcaPrezenta
End If

If Forms!prezenta!Implicit = False Then Call AfisareInitiala

Select Case CInt(Popup_Var1)
    Case 1 'adauga prezenta
        'Forms!prezenta!Plati.enabled = False
        Forms!prezenta!CautT.Enabled = False
        Forms!prezenta!CAUT.Enabled = False
        Forms!prezenta!CautB.Enabled = False
          
        Forms!prezenta!Prezenta_sub.Form.FormFooter.visible = True
        
        Forms!prezenta!Prezenta_sub!ZilePrezenta.Enabled = True
        Forms!prezenta!Prezenta_sub!CalT.Enabled = True
        Forms!prezenta!Prezenta_sub!Cal.Enabled = True
        Forms!prezenta!Prezenta_sub!Del.Enabled = True
        Forms!prezenta!Prezenta_sub!Calc.Enabled = False
        
        Forms!prezenta!Prezenta_sub!btnUndo.Enabled = True
        Forms!prezenta!Prezenta_sub!addc.Enabled = True
        Forms!prezenta!Prezenta_sub!sav.Enabled = True
        
    Case 2 'modifica prezenta
        Forms!prezenta!CautT.Enabled = False
        Forms!prezenta!CAUT.Enabled = False
        Forms!prezenta!CautB.Enabled = False
        
        Forms!prezenta!Prezenta_sub.Form.FormFooter.visible = True
        
        Forms!prezenta!Prezenta_sub!ZilePrezenta.Enabled = True
        Forms!prezenta!Prezenta_sub!CalT.Enabled = True
        Forms!prezenta!Prezenta_sub!Cal.Enabled = True
        Forms!prezenta!Prezenta_sub!Del.Enabled = True
        Forms!prezenta!Prezenta_sub!Calc.Enabled = False
        
        Forms!prezenta!Prezenta_sub!btnUndo.Enabled = True
        Forms!prezenta!Prezenta_sub!addc.Enabled = True
        Forms!prezenta!Prezenta_sub!sav.Enabled = True
    
    Case 3
        Forms!prezenta!CautT.Enabled = False
        Forms!prezenta!CAUT.Enabled = False
        Forms!prezenta!CautB.Enabled = False
        Call AfisarePlati
    
    Case 4
        Call AfisareRapoarte
        Forms!prezenta!Plati!DI = dhFirstDayInMonth(CDate("01/" & DLookup("Luna", "LunaD", "IDL=" & Forms!prezenta!IDL) & "/" & DLookup("Anul", "LunaD", "IDL=" & Forms!prezenta!IDL)))
        Forms!prezenta!Plati!DSF = dhLastDayInMonth(CDate("01/" & DLookup("Luna", "LunaD", "IDL=" & Forms!prezenta!IDL) & "/" & DLookup("Anul", "LunaD", "IDL=" & Forms!prezenta!IDL)))
        Forms!prezenta!CautT.Enabled = False
        Forms!prezenta!CAUT.Enabled = False
        Forms!prezenta!CautB.Enabled = False
    Case 5
        Forms!prezenta!CautT.Enabled = False
        Forms!prezenta!CAUT.Enabled = False
        Forms!prezenta!CautB.Enabled = False
        Call AfisareRetur
End Select

End Sub

Public Sub IncarcaPrezenta(Optional DoarAfiseaza As Boolean = False)

Dim RC As Recordset, TC As Recordset, Qc As DAO.QueryDef
Dim i As Integer
Dim DateNoi As Boolean: DateNoi = False

Set RC = OpenParamRecordset("SELECT Prezenta.*, SitLunara.SID, SitLunara.SIC, SitLunara.Platita, SitLunara.Retur FROM Prezenta LEFT JOIN SitLunara ON Prezenta.IDP = SitLunara.IDP WHERE CStr([IDL]) Like '" & Forms!prezenta!IDL & "' AND CStr([IDG])='" & Forms!prezenta!IDG & "'", 2)
Set TC = CurrentDb.OpenRecordset("Prezenta_L", 2)

Forms!prezenta!Prezenta_sub.SourceObject = ""

If DoarAfiseaza Then GoTo Afiseaza

GolesteTabel TC

If RC.EOF Then
    Call GolesteTabel(, "Prezenta_sub_L")
    If Nz(Forms!prezenta!IDG, -1) = -1 Then
        'MsgBox "Selecteaza grupa dorita pentru a introduce / modifica prezenta!", vbOKOnly + vbInformation
        GoTo Finish
    Else
        If Nz(Forms!prezenta!IDL, -1) = -1 Then
            'MsgBox "Selecteaza luna dorita pentru a introduce / modifica prezenta!", vbOKOnly + vbInformation
            GoTo Finish
        End If
        
        'If MsgBox("Nu exista date in luna " & Forms!prezenta!Luni!LunaT & "/" & Forms!prezenta!Luni!Anul & "!" & vbCrLf & "Doresti preluarea copiilor?", vbYesNo + vbQuestion) = vbNo Then GoTo Finish
        DateNoi = True
        
        Set RC = CurrentDb.OpenRecordset("SELECT * FROM Platitori WHERE CStr(IDG) like '" & Forms!prezenta!IDG & "'", 2)
        
        If RC.EOF Then
            MsgBox "Nu exista copii introdusi pentru grupa selectata!", vbOKOnly + vbInformation
            GoTo Finish
        Else
            Dim IDV As Integer, IDZ As Integer, ZileLuna As Integer, DI As Date, DSF As Date
            
            DI = DateValue("01/" & Forms!prezenta!Luna & "/" & Forms!prezenta!Anul)
            DSF = dhLastDayInMonth(DateValue("01/" & Forms!prezenta!Luna & "/" & Forms!prezenta!Anul))
            
            ZileLuna = Work_Days(DI, DSF)
            
            If Nz(ZileLuna, 0) = 0 Then
                MsgBox "Nu se pot introduce date fara numarul standard de zile!", vbOKOnly + vbCritical
                GoTo Finish
            End If
            
            IDZ = Nz(DMax("IDZ", "Prezenta_L"), Nz(DMax("IDZ", "Prezenta"), 0)) + 1
            
            POP RC
            
            For i = 0 To RC.RecordCount - 1
                TC.AddNew
                    TC!IDZ = IDZ + i
                    TC!IDV = Parent!IDV
                    TC!IDL = Forms!prezenta!IDL
                    TC!IDP = RC!IDP
                    TC!IDG = Forms!prezenta!IDG
                    TC!ZileLuna = ZileLuna
                TC.Update
                RC.MoveNext
            Next i
        End If
    End If
Else

Afiseaza:
    Call GolesteTabel(TC)
    
    If CBool(Nz(DLookup("Inchisa", "LunaD", "IDL=" & Nz(Forms!prezenta!IDL, 0)), False)) = True Then 'daca luna este blocata afisez datele salvate
        Set RC = CurrentDb.OpenRecordset("SELECT * FROM SS_Buget WHERE IDL=" & Forms!prezenta!IDL & " AND IDG = " & Forms!prezenta!IDG)
        If RC.EOF Then GoTo Finish
        
        Do While Not RC.EOF
            TC.AddNew
                TC!IDZ = RC!IDZ
                TC!IDP = RC!IDP
                TC!IDL = RC!IDL
                TC!IDG = RC!IDG
                TC!ZilePrezenta = RC!ZilePrezenta
                TC!ValoareContract = RC!ValoareContract
                TC!ValoareTotala = RC!ValoareTotala
                TC!Detalii = RC!Detalii
                TC!Restanta = RC!Restanta
                TC!SID = RC!SID
                TC!SIC = RC!SIC
                TC!Plata = RC!Plata
                TC!Retur = RC!Retur
            TC.Update
            RC.MoveNext
        Loop
        
    Else
        Do While Not RC.EOF
            TC.AddNew
                TC!IDZ = RC!IDZ
                TC!IDP = RC!IDP
                TC!IDL = RC!IDL
                TC!IDV = RC!IDV
                TC!IDG = RC!IDG
                TC!ZileLuna = RC!ZileLuna
                TC!ZilePrezenta = RC!ZilePrezenta
                TC!ZileAbsenta = RC!ZileAbsenta
                TC!MZCPrezenta = RC!MZCPrezenta
                TC!MZCAbsenta = RC!MZCAbsenta
                TC!ValoareContract = RC!ValoareContract
                TC!ValoareMancare = RC!ValoareMancare
                TC!ReducereFrate = RC!ReducereFrate
                TC!Detalii = RC!Detalii
                TC!ReducereBonus = RC!ReducereBonus
                TC!SI = RC!SI
                TC!Restanta = RC!Restanta
                TC!SID = RC!SID
                TC!SIC = RC!SIC
                TC!Plata = RC!Platita
                TC!Retur = RC!Retur
                TC!ValoareTotala = RC![ValoareContract] + RC![SID] + RC![Retur] - RC![Platita] - RC![SIC]
            TC.Update
            RC.MoveNext
        Loop
    End If
'    CurrentDb.Execute "UPDATE Prezenta_L INNER JOIN tmpSitDebPrezenta ON Prezenta_L.IDP = tmpSitDebPrezenta.IDP SET Prezenta_L.sid = [tmpSitDebPrezenta]![sid], Prezenta_L.sic = [tmpSitDebPrezenta]![sic], Prezenta_L.ValoareTotala = [prezenta_l]![valoarecontract]+[tmpSitDebPrezenta]![sid]+[tmpSitDebPrezenta]![Retur]-[tmpSitDebPrezenta]![Platita]-[tmpSitDebPrezenta]![sic], Prezenta_L.Plata = [tmpSitDebPrezenta]![Platita], Prezenta_L.Retur = [tmpSitDebPrezenta]![Retur]"
    
'    Set Rc = CurrentDb.OpenRecordset("SELECT * FROM Prezenta_sub WHERE IDZ IN (SELECT IDZ FROM Prezenta_L)", 2)
'    Set Tc = CurrentDb.OpenRecordset("SELECT * FROM Prezenta_Sub_L", 2)
    
'    Call GolesteTabel(Tc)
    
'    If Rc.EOF Then 's-a salvat luna fara sa se introduca prezenta
'        GoTo Finish
'    End If
    
'    POP Rc
    
'    Do While Not Rc.EOF
'        Tc.AddNew
'            Tc!IDX = Rc!IDX
'            Tc!IDZ = Rc!IDZ
'            Tc!IDL = Rc!IDL
'            Tc!DI = Rc!DI
'            Tc!DSF = Rc!DSF
'            Tc!nrzile = Rc!nrzile
'            Tc!absent = Rc!absent
'        Tc.Update
'        Rc.MoveNext
'    Loop
End If

Finish:

Forms!prezenta!Prezenta_sub.SourceObject = IIf(Nz(DLookup("VL", "CFGs", "F='XX' AND CFG='Platitor'"), "Buget") = "Buget", "Prezenta_sub_buget", "Prezenta_sub")

Set RC = Nothing
Set TC = Nothing

End Sub
