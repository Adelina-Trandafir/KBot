Function ChangeProperty(strPropName As String, varPropType As Variant, varPropValue As Variant) As Integer
    Dim Dbs As Object, prp As Variant
    Const conPropNotFoundError = 3270

    Set Dbs = CurrentDb
    On Error GoTo Change_Err
    Dbs.Properties(strPropName) = varPropValue
    ChangeProperty = True

Change_Bye:
    Exit Function

Change_Err:
    If Err = conPropNotFoundError Then    ' Property not found.
        Set prp = Dbs.CreateProperty(strPropName, _
            varPropType, varPropValue)
        Dbs.Properties.Append prp
        Resume Next
    Else
        ' Unknown error.
        ChangeProperty = False
        Resume Change_Bye
    End If
End Function

Function vercnp(CNP As String) As Integer
Dim Watch As Integer
If Len(CNP) <> 13 Then
    vercnp = 0
    Exit Function
End If


Watch = val(Mid(CNP, 4, 2))
If Watch > 12 Then
    vercnp = 2
    Exit Function
End If

If val(Mid(CNP, 6, 2)) > 31 Then
    vercnp = 3
    Exit Function
End If

If val(Mid(CNP, 8, 2)) > 52 Then
    vercnp = 4
    Exit Function
End If

If val(Right(CNP, 1)) <> UltimaCifra(CNP) Then
    vercnp = 5
    Exit Function
End If

vercnp = -1
End Function

End Function

Function fStrComp(Text1 As String, Text2 As String) As Long
Dim Pozitie As Double, Litera As Double, Valoare1 As Double, Valoare2 As Double

Text1 = UCase(Trim(Replace(Text1, " ", "")))

For i = 1 To Len(Text1)
    c = Mid(Text1, i, 1)
    Pozitie = i
    Litera = Asc(c)
    Valoare1 = Valoare1 + 10000 + Litera
Next i

'Text1 = Len(Text1) & "|" & Valoare

'==============

Text2 = UCase(Trim(Replace(Text2, " ", "")))

For i = 1 To Len(Text2)
    c = Mid(Text2, i, 1)
    Pozitie = i
    Litera = Asc(c)
    Valoare2 = Valoare2 + 10000 + Litera
Next i

'Text2 = Len(Text2) & "|" & Valoare

fStrComp = Abs(Valoare1 - Valoare2)

Function FindAutoNumber(TableName As String) As String
    Dim Dbs As DAO.Database
    Dim tdf As DAO.TableDef
    Dim fld As DAO.Field
    Set Dbs = CurrentDb
    Set tdf = Dbs.TableDefs(TableName)
    For Each fld In tdf.Fields

            FindAutoNumber = fld.Name

Public Function AddRecord(Tbl As String, Valori As String) As Boolean
Dim RC As Recordset
Dim i As Integer
Dim arrFld As Variant

arrFld = Split(Valori, ",")

If Not IsArray(arrFld) Then GoTo Final

Set RC = CurrentDb.OpenRecordset(Tbl, 2)

If UBound(arrFld) + 1 <> RC.Fields.Count Then
    MsgBox "Numarul de campuri nu corespunde!", vbOKOnly + vbCritical
    Set RC = Nothing
End If

RC.AddNew

For i = 0 To RC.Fields.Count - 1
    If arrFld(i) = "@" Then GoTo Pas
    
    If Left(arrFld(i), 1) = "'" Then
        If Right(arrFld(i), 1) = "'" Then
            RC.Fields(i).value = Replace(arrFld(i), "'", "")
            GoTo Pas
        Else
            MsgBox "Eroare formatare string (" & i & ")!", vbOKOnly + vbCritical
            GoTo Final
        End If
    End If
    
    If IsDate(arrFld(i)) Then
        RC.Fields(i).value = CDate(arrFld(i))
        GoTo Pas
    End If
    
    If IsNumeric(arrFld(i)) Then
        RC.Fields(i).value = CDbl(arrFld(i))
        GoTo Pas
    End If
    
Pas:
Next i

RC.Update

Set RC = Nothing

Final:

End Function

Public Function ActualizeazaCamp(Tbl As String, Camp As String, ValoareNoua As Variant, Optional Conditie As String = "") As Integer
''' 0 = Actualizare corecta
''' 1 = Lipsa tabel
''' 2 = Lipsa camp
''' 3 = Lipsa date (conditie eronata?)
''' 9 = Alta eroare

On Error GoTo Eroare

If Not ExistaTabel(Tbl) Then
    ActualizeazaCamp = 1
    Exit Function
End If

If Not ExistaCamp(Tbl, Camp) Then
    ActualizeazaCamp = 2
    Exit Function
End If

Dim RC As DAO.Recordset

If Nz(Conditie, "") <> "" Then
    Set RC = CurrentDb.OpenRecordset("SELECT * FROM [" & Tbl & "] WHERE " & Conditie & ";", 2)
Else
    Set RC = CurrentDb.OpenRecordset("SELECT * FROM [" & Tbl & "];", 2)
End If

If Not POP(RC) Then
    ActualizeazaCamp = 3
    Set RC = Nothing
    Exit Function
End If

Do While Not RC.EOF
    RC.Edit
        RC.Fields(Camp).value = ValoareNoua
    RC.Update
    
    RC.MoveNext
Loop

Set RC = Nothing

ActualizeazaCamp = 0

Exit Function

Eroare:
    DebugPrint Err.Description, "Modules/mdl_Functii_Generale/ActualizeazaCamp()"
    ActualizeazaCamp = 9
    If Not RC Is Nothing Then Set RC = Nothing
End Function

Function FormatN(Optional ByVal Suma_Text As String = "", Optional Lungime As Integer = 13, Optional Round1Cut2 As Byte = 1) As String
Dim vInt As String, vDec As String, vSuma As String, Negativ As Boolean, i As Integer
Dim dZec As Integer, dSep As String

If Suma_Text = "" Then
    FormatN = ""
    Exit Function
End If

If dZec = 0 Then
    If Nz(TempVars.Item("Zec"), "") = "" Then
        TempVars.Item("Zec") = Zec()
        dZec = Zec()
    Else
        dZec = TempVars.Item("Zec").value
    End If
End If

If dSep = "" Then
    If Nz(TempVars.Item("Sep"), "") = "" Then
        TempVars.Item("Sep") = New_DSep()
        dSep = New_DSep()
    Else
        dSep = TempVars.Item("Sep").value
    End If
End If

If Left(Suma_Text, 1) = "-" Then
    Negativ = True
    Suma_Text = Mid(Suma_Text, 2)
Else
    Negativ = False
End If

vInt = Split(Suma_Text, dSep)(0)

For i = 1 To CInt(Len(vInt) / 3)
    vSuma = "." & Right(vInt, 3) & vSuma
    vInt = Replace(vInt, Right(vInt, 3), "")
Next i

If vInt <> "" Then vSuma = vInt & vSuma 'pune cifrele ramase daca len(vint) / 3 are rest

If Left(vSuma, 1) = "." Then vSuma = Mid(vSuma, 2) 'scoate punctul daca len(vint) mod 3 = 0

If InStr(1, Suma_Text, dSep) Then
    If Round1Cut2 = 1 Then vDec = Split(Round("0" & dSep & Split(Suma_Text, dSep)(1), dZec), dSep)(1)   'pune zecimalele daca este cazul
    If Round1Cut2 = 2 Then vDec = Left(Split(Suma_Text, dSep)(1), dZec)
End If

vSuma = vSuma & "," & vDec & RepChar("0", dZec - Len(vDec))

If Negativ Then vSuma = "-" & vSuma

If Lungime < Len(vSuma) Then
    FormatN = vSuma
Else
    FormatN = Space(Lungime - Len(vSuma)) & vSuma
End If

End Function


Private Function RepChar(CH As String, Cn As Integer) As String
Dim i As Integer

For i = 1 To Cn
    RepChar = RepChar & CH
Next i
End Function
